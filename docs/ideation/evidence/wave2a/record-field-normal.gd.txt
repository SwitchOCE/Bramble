extends SceneTree
## Re-record complete normal games on the archived 0946909 content and current runtime.

const FrozenContent := preload("res://tests/core/field_frozen_content.gd")
const BASIS: String = "res://tests/fixtures/field/normal-0946909.json"
const CONTENT: String = "res://tests/fixtures/field/content-0946909.json"
var _fights: Array = []
var _keys: Array = []


func _initialize() -> void:
	var arguments: PackedStringArray = OS.get_cmdline_user_args()
	if arguments.size() != 2:
		printerr("Expected output path and current runtime commit")
		quit(2)
		return
	var frozen: FrozenContent = FrozenContent.new()
	frozen.setup(CONTENT)
	var basis: Dictionary = JSON.parse_string(FileAccess.get_file_as_string(BASIS))
	var games: Array[Dictionary] = []
	for previous: Dictionary in basis["games"]:
		var source: Dictionary = previous["config"]
		var config: GameConfig = FightCapture.config_for(source)
		# Pin every current resolved value, including capture precision/wording metadata.
		var pinned: Dictionary = config.resolved_values()
		_keys = pinned.keys()
		config = FightCapture.config_for(pinned)
		var session: GameSession = GameSession.create(
			CoreUtil.to_int(previous["seed"]), frozen.library, config
		)
		_fights.clear()
		session.fight_capture_observer = _remember
		while not session.is_over():
			assert(session.end_turn() == "")
		var game: Dictionary = {
			"fights": _fights.duplicate(true),
			"standings": session.get_standings(),
			"game_log": session.get_game_log()
		}
		var digest: String = FightCapture.canonical(game).sha256_text()
		games.append(
			{
				"seed": previous["seed"],
				"config": pinned,
				"fight_count": _fights.size(),
				"sha256": digest
			}
		)
		print(
			"field_rerecord ",
			JSON.stringify(
				{
					"seed": previous["seed"],
					"old_fight_count": previous["fight_count"],
					"new_fight_count": _fights.size(),
					"old_sha256": previous["sha256"],
					"new_sha256": digest
				}
			)
		)
	frozen.restore()
	var output: FileAccess = FileAccess.open(arguments[0], FileAccess.WRITE)
	assert(output != null)
	output.store_string(
		(
			JSON.stringify(
				{
					"source_commit": arguments[1],
					"content_source_commit": "0946909",
					"previous_fixture": BASIS,
					"games": games
				},
				"\t",
				true,
				true
			)
			+ "\n"
		)
	)
	output.close()
	quit(0)


func _remember(input: Dictionary, result: Dictionary, _combat: Combat) -> void:
	var normalized: Dictionary = input.duplicate(true)
	var config: Dictionary = normalized["config"]
	for key: String in config.keys():
		if key not in _keys:
			assert(
				GameConfig.defaults()._values.has(key) or Pass3Rates.defaults().has(key),
				"unknown config key: %s" % key
			)
			config.erase(key)
	_fights.append({"input": normalized, "result": result.duplicate(true)})
