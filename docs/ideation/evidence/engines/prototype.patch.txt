+class_name ProtoFlags
+extends RefCounted
+## Prototype switches for the engines design probe (ideation/c-engines). All off = current build.
+
+static var last_blow: bool = false
+static var formation: bool = false
+static var heirloom: bool = false
+static var last_stand: bool = false
+static var kin_only: bool = false
+static var stood: Dictionary = {}
+
+
+## BRAMBLE_PROTO=last_blow,last_four,formation,heirloom (or "pkg") switches them on for play.sh.
+static func from_env(config: GameConfig) -> void:
+	if not OS.has_environment("BRAMBLE_PROTO"):
+		return
+	var names: PackedStringArray = OS.get_environment("BRAMBLE_PROTO").split(",")
+	var all: bool = names.has("pkg")
+	last_blow = all or names.has("last_blow")
+	formation = all or names.has("formation")
+	heirloom = all or names.has("heirloom")
+	last_stand = names.has("last_stand")
+	kin_only = names.has("kin_only")
+	if all or names.has("last_four"):
+		config.set_value("damage_cap_off_at_alive", 4)
+
+
+## Neighbours of `slot` on `board` that share a tribe with `unit`, at most `most`.
+static func kin_touching(board: Array, slot: int, unit: BoardUnit, most: int = 3) -> int:
+	var count: int = 0
+	for other_slot: int in BoardGeometry.neighbours(slot):
+		var other: Variant = board[other_slot]
+		if other == null:
+			continue
+		var other_unit: BoardUnit = other
+		for tribe: Variant in unit.tribes():
+			if other_unit.has_tribe(str(tribe)):
+				count += 1
+				break
+	return mini(count, most)
+
+
+## Prototype Last Stand: in the last four, the first loss that would knock a hero out from above
+## 5 Health leaves it on 1. Once a game per hero.
+static func take(player: PlayerState, damage: int, living: int) -> void:
+	var key: int = player.get_instance_id()
+	var after: int = player.health - maxi(0, damage - player.armor)
+	if last_stand and living <= 4 and player.health > 5 and after <= 0 and not stood.has(key):
+		stood[key] = true
+		player.armor = 0
+		player.health = 1
+		return
+	HeroRules.take_hero_damage(player, damage)
